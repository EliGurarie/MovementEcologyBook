# helper function
plot.fpt <- function (x, radii, main = "", ...) {
  if (!inherits(x, "fipati")) 
    stop("x should be of class 'fipati'")
  
  for(i in 1:length(radii))
  {
    u <- x[[1]]
    plot(attr(u, "date"), u[, i], 
         ylab = "FPT", main=paste(main, radii[i], sep=""), ...)
    lines(attr(u, "date"), u[, i])
  }
}
